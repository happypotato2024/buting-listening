cmake_minimum_required(VERSION 3.22)
project(buting_lab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_HTTPLIB OFF CACHE BOOL "" FORCE)
if(ENGINE STREQUAL "whisper")
  add_subdirectory("${ENGINE_SOURCE}" engine)
  add_library(buting_whisper SHARED whisper_jni.cpp)
  target_link_libraries(buting_whisper PRIVATE whisper log)
  target_link_options(buting_whisper PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
else()
  add_subdirectory("${ENGINE_SOURCE}" engine)
  add_library(buting_translate SHARED llama_jni.cpp)
  target_link_libraries(buting_translate PRIVATE llama log)
  target_link_options(buting_translate PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")
endif()
